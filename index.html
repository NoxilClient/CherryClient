pub fn build_chunk_mesh_parallel(sections: &[ChunkSection]) -> Vec<PackedVertex> {
    sections
        .par_iter() // Lock-free work-stealing across all CPU cores
        .filter(|sec| !sec.is_empty())
        .flat_map(|sec| {
            let mut greedy_quads = GreedyMesher::with_simd_culling();
            greedy_quads.ao_lightmap_pass(sec);
            greedy_quads.pack_vertices_32bit() // 4 bytes per vertex vs 32 bytes in Vanilla
        })
        .collect()
}
